// Array to store our created servers locally let servers = JSON.parse(localStorage.getItem('flavo_servers') || '[]'); let activeServerIndex = null; // DOM Elements const sidebarList = document.getElementById('sidebar-list'); const emptyState = document.getElementById('empty-state'); const creatorView = document.getElementById('creator-view'); const serverView = document.getElementById('server-view'); function saveServers() { localStorage.setItem('flavo_servers', JSON.stringify(servers)); } // Functions to Switch Views function openCreatorView(pushHistory = true) { if (pushHistory) { window.history.pushState({}, '', '/flavosmp/link-creator/'); } creatorView.classList.add('active'); serverView.classList.remove('active'); document.querySelectorAll('.server-item').forEach(el => el.classList.remove('active')); document.getElementById('serverNameInput').value = ''; } function openServerView(index, pushHistory = true) { activeServerIndex = index; const srv = servers[index]; const slug = srv.name.toLowerCase().replace(/[^a-z0-9]+/g, '-'); if (pushHistory) { window.history.pushState({}, '', `/flavosmp/link-creator/${slug}`); } document.getElementById('activeServerTitle').innerText = srv.name; document.getElementById('activeCommandText').innerText = `/server-link ${srv.code}`; creatorView.classList.remove('active'); serverView.classList.add('active'); updateSidebar(); } // --- REAL PIPEDREAM CREATION LOGIC --- async function simulateCreation() { const input = document.getElementById('serverNameInput'); const name = input.value.trim(); if(!name) return alert("Please enter a server name!"); const btn = document.querySelector('#creator-view .btn-fire'); const originalText = btn.innerText; btn.innerText = "GENERATING..."; btn.disabled = true; try { const response = await fetch("https://eoed3phjmlklizm.m.pipedream.net/", { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'create_link', serverName: name }) }); const data = await response.json(); if (data.success) { servers.push({ name: name, code: data.code, // Real database code from Pipedream status: 'Awaiting Discord Link...' }); saveServers(); openServerView(servers.length - 1); } else { alert("Error: " + (data.message || "Unknown error")); } } catch (err) { alert("Network error. Could not reach Pipedream."); } finally { btn.innerText = originalText; btn.disabled = false; } } // --- REAL PIPEDREAM REGENERATE LOGIC --- async function generateNewCodeForActive() { if(activeServerIndex === null) return; const srv = servers[activeServerIndex]; const btn = document.querySelector('#server-view .btn-fire'); const originalText = btn.innerText; btn.innerText = "GENERATING..."; btn.disabled = true; try { const response = await fetch("https://eoed3phjmlklizm.m.pipedream.net/", { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'create_link', serverName: srv.name }) }); const data = await response.json(); if (data.success) { servers[activeServerIndex].code = data.code; saveServers(); document.getElementById('activeCommandText').innerText = `/server-link ${data.code}`; } else { alert("Error: " + (data.message || "Unknown error")); } } catch (err) { alert("Network error. Could not reach Pipedream."); } finally { btn.innerText = originalText; btn.disabled = false; } } function updateSidebar() { if (servers.length === 0) { emptyState.style.display = 'block'; return; } emptyState.style.display = 'none'; Array.from(sidebarList.children).forEach(child => { if(child.id !== 'empty-state') child.remove(); }); servers.forEach((srv, index) => { const div = document.createElement('div'); div.className = `server-item ${index === activeServerIndex ? 'active' : ''}`; div.onclick = () => openServerView(index); div.innerHTML = `

${srv.name}

${srv.status}

`; sidebarList.appendChild(div); }); } // --- ROUTING LOGIC ON PAGE LOAD --- function handleRoutingOnLoad() { const urlParams = new URLSearchParams(window.location.search); const redirectPath = urlParams.get('redirect'); if (redirectPath) { window.history.replaceState({}, '', redirectPath); } const currentPath = window.location.pathname; if (currentPath.endsWith('/link-creator/') || currentPath.endsWith('/link-creator')) { openCreatorView(false); } else { const slug = currentPath.split('/').filter(Boolean).pop(); const serverIndex = servers.findIndex(srv => { const srvSlug = srv.name.toLowerCase().replace(/[^a-z0-9]+/g, '-'); return srvSlug === slug; }); if (serverIndex !== -1) { openServerView(serverIndex, false); } else { openCreatorView(false); } } } updateSidebar(); handleRoutingOnLoad();
